ChatGPT APIで画像プロンプトを自動生成（Python）

python 
import openai

client = openai.OpenAI(api_key="YOUR_API_KEY")

def generate_image_prompt(
    product_name: str,
    features: str,
    mood: str,
    size: str = "horizontal"
) -> str:
    """
    商品情報からChatGPT画像生成用の英語プロンプトを自動生成する
    mood: "professional" | "cute" | "natural" | "luxury"
    size: "horizontal" | "square" | "vertical"
    """

    system_prompt = """あなたは商品画像生成の専門プロンプトエンジニアです。
商品情報を元に、DALL-E 3 / gpt-image-1 で最高品質の商品画像を
生成するための英語プロンプトを作成してください。

プロンプトは以下の要素を含めてください:
- 商品の視覚的特徴
- 背景・環境設定
- ライティング指定
- カメラアングル・距離感
- 全体のムード・スタイル
- 技術的品質指定（4K, photorealistic 等）

プロンプトのみを返してください（説明文不要）。
"""

    user_msg = f"""
商品名: {product_name}
特徴: {features}
雰囲気: {mood}
バナーサイズ: {size}

上記の商品の魅力が最大限伝わる商品画像用プロンプトを
英語で生成してください。
"""

    response = client.chat.completions.create(
        model="gpt-4o-mini",
        messages=[
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": user_msg}
        ],
        max_tokens=300
    )
    return response.choices[0].message.content.strip()

# 使用例
prompt = generate_image_prompt(
    product_name="オーガニックフェイスクリーム",
    features="天然成分100%・無添加・敏感肌向け",
    mood="natural"
)
print(prompt)

gpt-image-1 で商品画像を生成（Python）

python 
import openai
import requests
from pathlib import Path
from datetime import datetime

client = openai.OpenAI(api_key="YOUR_API_KEY")

def generate_product_image(
    prompt: str,
    size: str = "1792x1024",
    output_dir: str = "./output"
) -> str:
    """
    gpt-image-1 で商品画像を生成してローカルに保存
    size options:
      横長バナー: "1792x1024"
      正方形:     "1024x1024"
      縦長:       "1024x1792"
    """

    response = client.images.generate(
        model="gpt-image-1",
        prompt=prompt,
        n=1,
        size=size,
        quality="high"
    )

    image_url = response.data[0].url

    # 画像をダウンロードして保存
    Path(output_dir).mkdir(parents=True, exist_ok=True)
    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    output_path = f"{output_dir}/banner_{size}_{timestamp}.png"

    img_data = requests.get(image_url).content
    Path(output_path).write_bytes(img_data)

    print(f"生成完了: {output_path}")
    return output_path


def batch_generate_banners(prompt: str, output_dir: str = "./banners"):
    """3パターンのバナーサイズで一括生成"""
    sizes = {
        "horizontal": "1792x1024",
        "square":     "1024x1024",
        "vertical":   "1024x1792",
    }
    results = {}
    for name, size in sizes.items():
        print(f"Generating {name} ({size})...")
        results[name] = generate_product_image(prompt, size, output_dir)
    return results


# バッチ実行例
if __name__ == "__main__":
    my_prompt = """
    Luxury organic face cream product photography.
    Elegant white jar on marble surface with fresh flowers.
    Soft natural lighting, bokeh background, minimalist style.
    4K photorealistic, commercial quality.
    """
    results = batch_generate_banners(my_prompt)
    print("全サイズ生成完了:", results)
