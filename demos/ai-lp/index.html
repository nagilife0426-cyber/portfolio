ChatGPT API でLP構成案を生成（Python）

python 
import openai
import json

client = openai.OpenAI(api_key="YOUR_API_KEY")

def generate_lp_structure(industry: str, target: str, appeal: str) -> dict:
    """ChatGPT APIでLP構成案を生成する"""

    prompt = f"""
あなたはLP（ランディングページ）設計の専門家です。
以下の情報を元に、最適なLP構成案をJSON形式で返してください。

業種: {industry}
ターゲット: {target}
訴求ポイント: {appeal}

以下のJSONフォーマットで返してください:
{{
  "sections": [
    {{
      "name": "セクション名",
      "purpose": "目的",
      "content_hint": "盛り込むべき内容のヒント"
    }}
  ],
  "hero_copy": "ヒーローキャッチコピー案",
  "cta_text": "CTAボタンテキスト"
}}
"""

    response = client.chat.completions.create(
        model="gpt-4o-mini",
        messages=[
            {"role": "system", "content": "LP設計の専門家として回答してください。"},
            {"role": "user", "content": prompt}
        ],
        response_format={"type": "json_object"}
    )

    return json.loads(response.choices[0].message.content)

# 実行例
result = generate_lp_structure(
    industry="税理士事務所",
    target="確定申告に悩む個人事業主",
    appeal="初回無料相談・オンライン対応・当日対応可"
)
print(json.dumps(result, ensure_ascii=False, indent=2))

Genspark API でビジュアル生成（Python）

python 
import requests
import base64
from pathlib import Path

def generate_lp_visual(
    industry: str,
    tone: str,
    output_path: str = "lp_visual.png"
) -> str:
    """Genspark APIでLP用ビジュアルを生成する"""

    # GensparkのImage Generation API (v1)
    url = "https://api.genspark.ai/v1/images/generate"
    headers = {
        "Authorization": f"Bearer YOUR_GENSPARK_API_KEY",
        "Content-Type": "application/json"
    }

    # プロンプトを業種・トーンに合わせて構築
    prompt = f"""
Professional landing page hero image for {industry}.
Style: {tone}, modern, minimalist.
High quality, web-ready, 16:9 ratio.
No text overlays. Clean background.
"""

    payload = {
        "prompt": prompt,
        "size": "1792x1024",
        "quality": "hd",
        "n": 1
    }

    response = requests.post(url, headers=headers, json=payload)
    response.raise_for_status()

    data = response.json()
    image_url = data["data"][0]["url"]

    # 画像をローカルに保存
    img_response = requests.get(image_url)
    Path(output_path).write_bytes(img_response.content)
    print(f"ビジュアル保存完了: {output_path}")
    return output_path

// スプレッドシートのA列に業種、B列にターゲット、C列に訴求を入力
// 実行するとD列にLP構成案JSONが書き込まれる

function generateLPStructures() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getActiveSheet();
  const data = sheet.getDataRange().getValues();
  const apiKey = PropertiesService.getScriptProperties().getProperty('OPENAI_API_KEY');

  for (let i = 1; i < data.length; i++) {
    const [industry, target, appeal] = data[i];
    if (!industry) continue;

    const prompt = `業種:${industry} ターゲット:${target} 訴求:${appeal}
のLP構成案をJSON形式で生成してください。`;

    const response = UrlFetchApp.fetch('https://api.openai.com/v1/chat/completions', {
      method: 'post',
      contentType: 'application/json',
      headers: { Authorization: `Bearer ${apiKey}` },
      payload: JSON.stringify({
        model: 'gpt-4o-mini',
        messages: [{ role: 'user', content: prompt }],
        response_format: { type: 'json_object' }
      })
    });

    const result = JSON.parse(response.getContentText());
    sheet.getRange(i + 1, 4).setValue(result.choices[0].message.content);
    Utilities.sleep(1000); // レート制限対策
  }
  SpreadsheetApp.getUi().alert('LP構成案の生成が完了しました');
}
