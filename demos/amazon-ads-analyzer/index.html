// Amazon広告レポート自動取得 & スプレッドシート記録 // 毎朝9時に自動実行（トリガー設定済み） const CONFIG = { SPREADSHEET_ID: 'YOUR_SPREADSHEET_ID', SHEET_NAME: '広告レポート', SLACK_WEBHOOK: 'https://hooks.slack.com/services/xxx', CLAUDE_API_KEY: PropertiesService.getScriptProperties().getProperty('CLAUDE_KEY'), }; function fetchAmazonReport() { const ss = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID); const sheet = ss.getSheetByName(CONFIG.SHEET_NAME); // Amazonレポート取得（Advertising API経由） const report = getAdvertisingReport(); // KPI計算 const kpi = { acos: (report.adSpend / report.adSales * 100).toFixed(1), roas: (report.adSales / report.adSpend * 100).toFixed(0), tacos: (report.adSpend / report.totalSales * 100).toFixed(1), date: new Date().toLocaleDateString('ja-JP'), }; // スプレッドシートに記録 sheet.appendRow([ kpi.date, kpi.acos, kpi.roas, kpi.tacos, report.clicks, report.orders, report.adSpend, report.adSales ]); // Claude APIでAI分析・改善提案を生成 const insight = generateClaudeInsight(kpi, report.keywords); // Slack通知送信 sendSlackNotification(kpi, insight); Logger.log(`レポート記録完了: ACOS ${kpi.acos}% / ROAS ${kpi.roas}%`); } function generateClaudeInsight(kpi, keywords) { const payload = { model: 'claude-opus-4-5', max_tokens: 500, messages: [{ role: 'user', content: `Amazon広告KPIを分析してください。 ACOS: ${kpi.acos}% / ROAS: ${kpi.roas}% / TACoS: ${kpi.tacos}% 上位キーワード: ${JSON.stringify(keywords.slice(0,5))} 改善提案を3点、簡潔にリストアップしてください。` }] }; const res = UrlFetchApp.fetch('https://api.anthropic.com/v1/messages', { method: 'post', contentType: 'application/json', headers: { 'x-api-key': CONFIG.CLAUDE_API_KEY, 'anthropic-version': '2023-06-01' }, payload: JSON.stringify(payload), }); return JSON.parse(res.getContentText()).content[0].text; }
