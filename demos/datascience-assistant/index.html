S3バケット作成 & ライフサイクル設定
AWS CLI / boto3 
import boto3
import json

# S3クライアント初期化
s3 = boto3.client('s3', region_name='ap-northeast-1')

def create_data_bucket(bucket_name: str) -> dict:
    """データレイク用バケットを作成してライフサイクル設定"""

    # バケット作成
    s3.create_bucket(
        Bucket=bucket_name,
        CreateBucketConfiguration={'LocationConstraint': 'ap-northeast-1'}
    )

    # バージョニング有効化
    s3.put_bucket_versioning(
        Bucket=bucket_name,
        VersioningConfiguration={'Status': 'Enabled'}
    )

    # ライフサイクル設定（30日後にIA、90日後にGlacier）
    lifecycle_config = {
        'Rules': [{
            'ID': 'DataArchiveRule',
            'Status': 'Enabled',
            'Transitions': [
                {'Days': 30, 'StorageClass': 'STANDARD_IA'},
                {'Days': 90, 'StorageClass': 'GLACIER'}
            ]
        }]
    }
    s3.put_bucket_lifecycle_configuration(
        Bucket=bucket_name,
        LifecycleConfiguration=lifecycle_config
    )

    print(f"✅ バケット作成完了: {bucket_name}")
    return {'bucket': bucket_name, 'status': 'created'}

# 実行例
create_data_bucket("my-company-datalake-prod")

Lambda 関数デプロイ（ZIP自動パッケージング）
AWS Lambda 
import boto3, zipfile, os, json

def deploy_lambda(
    function_name: str,
    handler_path: str,
    role_arn: str,
    env_vars: dict = {}
) -> dict:
    """Lambdaをパッケージングしてデプロイ or 更新"""

    lambda_client = boto3.client('lambda', region_name='ap-northeast-1')

    # ZIPパッケージ作成
    zip_path = f"/tmp/{function_name}.zip"
    with zipfile.ZipFile(zip_path, 'w') as z:
        z.write(handler_path, 'lambda_function.py')

    with open(zip_path, 'rb') as f:
        zip_bytes = f.read()

    # 既存チェック → 作成 or 更新
    try:
        lambda_client.get_function(FunctionName=function_name)
        # 既存: コード更新
        res = lambda_client.update_function_code(
            FunctionName=function_name,
            ZipFile=zip_bytes
        )
        action = "updated"
    except lambda_client.exceptions.ResourceNotFoundException:
        # 新規作成
        res = lambda_client.create_function(
            FunctionName=function_name,
            Runtime='python3.12',
            Role=role_arn,
            Handler='lambda_function.handler',
            Code={'ZipFile': zip_bytes},
            Timeout=30,
            Environment={'Variables': env_vars}
        )
        action = "created"

    print(f"✅ Lambda {action}: {function_name}")
    return {'function': function_name, 'action': action}

deploy_lambda("data-processor", "./handler.py", "arn:aws:iam::123456789:role/LambdaRole")

CloudWatch アラート一括設定
CloudWatch 
import boto3

cw = boto3.client('cloudwatch', region_name='ap-northeast-1')

# 設定したいアラートの定義
ALERTS = [
    {
        'name': 'Lambda-ErrorRate-High',
        'namespace': 'AWS/Lambda',
        'metric': 'Errors',
        'threshold': 5,
        'period': 300,
        'description': 'Lambda エラー率が5件/5分超'
    },
    {
        'name': 'RDS-CPU-High',
        'namespace': 'AWS/RDS',
        'metric': 'CPUUtilization',
        'threshold': 80,
        'period': 60,
        'description': 'RDS CPU 80%超'
    },
    {
        'name': 'S3-4xxErrors-High',
        'namespace': 'AWS/S3',
        'metric': '4xxErrors',
        'threshold': 100,
        'period': 300,
        'description': 'S3 4xxエラー多発'
    }
]

def setup_alerts(sns_topic_arn: str):
    for alert in ALERTS:
        cw.put_metric_alarm(
            AlarmName=alert['name'],
            AlarmDescription=alert['description'],
            Namespace=alert['namespace'],
            MetricName=alert['metric'],
            Threshold=alert['threshold'],
            Period=alert['period'],
            EvaluationPeriods=1,
            ComparisonOperator='GreaterThanThreshold',
            Statistic='Sum',
            AlarmActions=[sns_topic_arn],
            TreatMissingData='notBreaching'
        )
        print(f"✅ アラート設定: {alert['name']}")

setup_alerts("arn:aws:sns:ap-northeast-1:123456789:alerts")
