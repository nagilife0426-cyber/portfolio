// Google Apps Script - フォーム自動処理
// Googleフォームの「送信時」トリガーに設定

function onFormSubmit(e) {
  const responses = e.namedValues;
  const name    = responses['お名前'][0];
  const email   = responses['メールアドレス'][0];
  const company = responses['会社名'][0] || '未入力';
  const type    = responses['お問い合わせ種別'][0];
  const message = responses['お問い合わせ内容'][0];
  const now     = new Date();

  // ① スプレッドシートに記録
  recordToSpreadsheet({ name, email, company, type, message, now });

  // ② 自動返信メール送信
  sendAutoReply({ name, email, type });

  // ③ Slack通知
  notifySlack({ name, company, type, message });
}

function recordToSpreadsheet({ name, email, company, type, message, now }) {
  const sheet = SpreadsheetApp
    .getActiveSpreadsheet()
    .getSheetByName('お問い合わせ一覧');

  sheet.appendRow([
    Utilities.formatDate(now, 'Asia/Tokyo', 'yyyy/MM/dd HH:mm'),
    name, email, company, type, message, '未対応'
  ]);
}

function sendAutoReply({ name, email, type }) {
  const templates = {
    '料金確認': `料金表をお送りしますので、ご確認ください。`,
    '納期確認': `ご要件をお聞かせいただいた後、お見積もりいたします。`,
    '技術相談': `詳細についてヒアリングのお時間をいただけますか。`,
    'その他':   `内容を確認の上、担当者よりご連絡いたします。`
  };

  const body = [
    `${name} 様`,
    ``,
    `お問い合わせありがとうございます。`,
    `【${type}】のご質問について：`,
    templates[type] || templates['その他'],
    ``,
    `通常1営業日以内にご連絡いたします。`,
    ``,
    `なぎ ワークス`
  ].join('\n');

  GmailApp.sendEmail(email, `【自動返信】${type}のお問い合わせを受け付けました`, body);
}

function notifySlack({ name, company, type, message }) {
  const webhookUrl = PropertiesService
    .getScriptProperties()
    .getProperty('SLACK_WEBHOOK_URL');

  const payload = {
    text: `📬 新しいお問い合わせ`,
    attachments: [{
      color: '#00d4ff',
      fields: [
        { title: '名前', value: name, short: true },
        { title: '会社', value: company, short: true },
        { title: '種別', value: type, short: true },
        { title: '内容', value: message.slice(0, 100) }
      ]
    }]
  };

  UrlFetchApp.fetch(webhookUrl, {
    method: 'post',
    contentType: 'application/json',
    payload: JSON.stringify(payload)
  });
}

記録・返信・通知
シート記録＋自動返信メール＋Slack通知
  • Google Apps Script（GAS）の onFormSubmit トリガーで、フォーム送信と同時にプログラムが自動起動します。
  • 受信内容をスプレッドシートに自動記録し、種別・部署・ステータスで仕分けできる形に整えます。
  • 送信者へ自動返信メールを送り、同時にSlackへ通知。担当者の取りこぼしを防ぎます。
  • すべてGoogleアカウント内で完結するため月額費用ゼロ。返信文面や通知先は自由に設定できます。
