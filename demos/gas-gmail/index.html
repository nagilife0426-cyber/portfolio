// Google Apps Script - Gmail 自動ラベル振り分け
// 毎時トリガーで実行 → 未処理メールを自動分類

const LABEL_RULES = [
  {
    label: '注文確認',
    color: 'sage',
    matchers: [
      { field: 'subject', contains: ['注文', 'ご注文', 'order', '購入完了'] },
      { field: 'from',    contains: ['amazon.co.jp', 'rakuten.co.jp'] }
    ]
  },
  {
    label: '問い合わせ',
    color: 'cyan',
    matchers: [
      { field: 'subject', contains: ['お問い合わせ', 'ご質問', '相談', '確認'] }
    ]
  },
  {
    label: '請求書',
    color: 'yellow',
    matchers: [
      { field: 'subject', contains: ['請求', 'invoice', '領収'] }
    ]
  },
  {
    label: '営業メール',
    color: 'red',
    matchers: [
      { field: 'subject', contains: ['ご提案', 'セール', '特別価格', 'PR'] },
      { field: 'body',    contains: ['この度は弊社サービスをご紹介'] }
    ]
  }
];

function autoLabelEmails() {
  // 過去24時間の未読メールを取得
  const threads = GmailApp.search(
    'is:unread -label:処理済み newer_than:1d',
    0, 50
  );

  let processed = 0;
  const stats = {};

  threads.forEach(thread => {
    const msg = thread.getMessages()[0];
    const subject = msg.getSubject();
    const from    = msg.getFrom();
    const body    = msg.getPlainBody().slice(0, 500);

    let matched = false;
    for (const rule of LABEL_RULES) {
      for (const m of rule.matchers) {
        const target = m.field === 'subject' ? subject
                     : m.field === 'from'    ? from : body;
        const hit = m.contains.some(
          kw => target.toLowerCase().includes(kw.toLowerCase())
        );
        if (hit) {
          applyLabel(thread, rule.label);
          stats[rule.label] = (stats[rule.label] || 0) + 1;
          matched = true;
          break;
        }
      }
      if (matched) break;
    }

    // マッチしなければ「要確認」ラベルを付与
    if (!matched) {
      applyLabel(thread, '要確認');
      stats['要確認'] = (stats['要確認'] || 0) + 1;
    }

    applyLabel(thread, '処理済み');
    processed++;
  });

  // 集計をスプレッドシートに記録
  if (processed > 0) {
    logStats(processed, stats);
  }
}

function applyLabel(thread, labelName) {
  let label = GmailApp.getUserLabelByName(labelName);
  if (!label) label = GmailApp.createLabel(labelName);
  thread.addLabel(label);
}

function logStats(count, stats) {
  const sheet = SpreadsheetApp
    .getActiveSpreadsheet()
    .getSheetByName('仕分けログ');
  const row = [new Date(), count, JSON.stringify(stats)];
  sheet.appendRow(row);
}

// 毎時実行トリガーを設定
function setHourlyTrigger() {
  ScriptApp.newTrigger('autoLabelEmails')
    .timeBased()
    .everyHours(1)
    .create();
}
