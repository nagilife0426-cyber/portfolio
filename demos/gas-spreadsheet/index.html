// Google Apps Script - 月次売上レポート自動生成
// スプレッドシートのツールバーから実行

function generateMonthlyReport() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const dataSheet = ss.getSheetByName('売上データ');
  const reportSheet = ss.getSheetByName('月次レポート')
    || ss.insertSheet('月次レポート');

  const data = dataSheet.getDataRange().getValues();
  const headers = data[0];
  const rows = data.slice(1).filter(r => r[0]); // 空行を除外

  // 集計処理
  let totalSales = 0;
  const byPerson = {};
  const byCategory = {};
  const byDate = {};

  rows.forEach(row => {
    const [date, product, category, qty, price, person] = row;
    const amount = qty * price;
    totalSales += amount;

    // 担当者別
    byPerson[person] = (byPerson[person] || 0) + amount;

    // カテゴリ別
    byCategory[category] = (byCategory[category] || 0) + amount;

    // 日付別
    const dateStr = Utilities.formatDate(
      new Date(date), 'Asia/Tokyo', 'MM/dd'
    );
    byDate[dateStr] = (byDate[dateStr] || 0) + amount;
  });

  // 担当者ランキング（降順ソート）
  const ranking = Object.entries(byPerson)
    .sort((a, b) => b[1] - a[1])
    .slice(0, 3);

  // レポートシートに書き出し
  reportSheet.clearContents();

  const now = Utilities.formatDate(
    new Date(), 'Asia/Tokyo', 'yyyy年M月度'
  );
  reportSheet.getRange(1, 1).setValue(`月次レポート - ${now}`);
  reportSheet.getRange(2, 1).setValue('総売上');
  reportSheet.getRange(2, 2).setValue(totalSales);
  reportSheet.getRange(3, 1).setValue('件数');
  reportSheet.getRange(3, 2).setValue(rows.length);

  // ランキング書き出し
  reportSheet.getRange(5, 1).setValue('担当者ランキング');
  ranking.forEach(([ name, amount ], i) => {
    reportSheet.getRange(6 + i, 1).setValue(`${i+1}位: ${name}`);
    reportSheet.getRange(6 + i, 2).setValue(amount);
  });

  // 完了通知
  SpreadsheetApp.getUi().alert(
    `✅ レポート生成完了\n総売上: ¥${totalSales.toLocaleString()}`
  );
}

// 毎月1日 09:00 に自動実行するトリガーを設定
function setMonthlyTrigger() {
  ScriptApp.newTrigger('generateMonthlyReport')
    .timeBased()
    .onMonthDay(1)
    .atHour(9)
    .create();
  Logger.log('毎月1日 9時のトリガーを設定しました');
}
