💡 GAS × Claude API の組み合わせが最強な理由
GAS（Google Apps Script）はスプレッドシート・Gmail・カレンダー・フォームと直接連携できるため、既存のGoogle Workspaceをそのまま使えます。そこにClaude APIを組み合わせることで、問い合わせ分類・見積文章生成・優先度判定など「AIでしかできない部分」を追加。新しいツールへの移行コストゼロで自動化できるのが最大のメリットです。

auto_input.gs — Googleフォーム問い合わせの自動入力・Slack通知 
// Googleフォームの送信をトリガーとしてスプレッドシートへ自動入力
// GAS（Google Apps Script）で実装 — 新規インストール不要

const SHEET_NAME = "案件管理";
const SLACK_WEBHOOK = "YOUR_SLACK_WEBHOOK_URL";

// フォーム送信時に自動実行されるトリガー関数
function onFormSubmit(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAME);
  const res = e.namedValues;  // フォームの回答

  // スプレッドシートに新しい行を追加
  const now = new Date();
  sheet.appendRow([
    Utilities.formatDate(now, "JST", "yyyy/MM/dd HH:mm"),  // 受付日時
    res["お名前"]?.[0] ?? "",                              // 顧客名
    res["電話番号"]?.[0] ?? "",                            // 連絡先
    res["ご依頼内容"]?.[0] ?? "",                          // 依頼内容
    res["サービス種別"]?.[0] ?? "",                        // リフォーム/クリーニング等
    "新規問い合わせ",                                         // ステータス
    "",                                                       // 担当者（後から入力）
  ]);

  // Slackに即時通知
  const message = {
    text: `🔔 *新規問い合わせ*\n`
        + `お名前: ${res["お名前"]?.[0]}\n`
        + `内容: ${res["ご依頼内容"]?.[0]}\n`
        + `種別: ${res["サービス種別"]?.[0]}`
  };
  UrlFetchApp.fetch(SLACK_WEBHOOK, {
    method: "post",
    contentType: "application/json",
    payload: JSON.stringify(message),
  });

  Logger.log("✅ 自動入力 + Slack通知完了");
}
reminder.gs — 返信遅延・見積未送付の自動検知・リマインド 
// 毎朝9時に自動実行して、アクションが必要な案件を検知してSlack通知
// cron設定: GASのトリガーで「毎日 9:00〜9:30」に設定

const REMIND_DAYS = 2;    // N日返信なしでアラート
const ESTIMATE_DAYS = 1;  // 見積書をN日以内に送付するルール
const BEFORE_DAYS = 7;    // 施工N日前にリマインド

function dailyCheck() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet()
                               .getSheetByName("案件管理");
  const data = sheet.getDataRange().getValues();
  const now = new Date();
  const alerts = [];

  // ヘッダー行をスキップして各案件を確認
  for (let i = 1; i < data.length; i++) {
    const [recvAt, name, tel, desc, type, status, staff, estimateAt, scheduleAt] = data[i];
    if (["完了", "辞退", "キャンセル"].includes(status)) continue;

    const diffDays = (d) => Math.floor((now - new Date(d)) / 86400000);

    // 返信遅延チェック
    if (recvAt && diffDays(recvAt) >= REMIND_DAYS && status === "新規問い合わせ") {
      alerts.push(`🚨 *${name}様* — 受付から${diffDays(recvAt)}日経過（未対応）`);
    }
    // 見積未送付チェック
    if (status === "見積作成中" && recvAt && diffDays(recvAt) > ESTIMATE_DAYS) {
      alerts.push(`⚠️ *${name}様* — 見積書 未送付 (${diffDays(recvAt)}日経過)`);
    }
    // 施工日リマインド
    if (scheduleAt) {
      const daysLeft = -diffDays(scheduleAt);
      if (daysLeft > 0 && daysLeft <= BEFORE_DAYS) {
        alerts.push(`📅 *${name}様* — 施工まで残${daysLeft}日`);
      }
    }
  }

  if (alerts.length > 0) {
    UrlFetchApp.fetch(SLACK_WEBHOOK, {
      method: "post", contentType: "application/json",
      payload: JSON.stringify({ text: "🔔 *要対応案件*\n\n" + alerts.join("\n") })
    });
  }
  Logger.log(`チェック完了: ${alerts.length}件のアラート`);
}
estimate_ai.gs — Claude API で見積書文章を自動生成 ← 差別化ポイント 
// GASからClaude APIを呼び出して見積書の文章を自動生成
// スプレッドシートのボタンを押すだけで見積書ドラフトが完成

const CLAUDE_API_KEY = "YOUR_ANTHROPIC_API_KEY";
const CLAUDE_MODEL   = "claude-opus-4-5";

function generateEstimate() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  const row = sheet.getActiveRange().getRow();
  const [_, name, tel, desc, type] = sheet.getRange(row, 1, 1, 5).getValues()[0];

  const prompt = `
あなたは住宅サービス会社のベテラン営業担当です。
以下の情報をもとに、丁寧でわかりやすい見積書の文章を作成してください。

【顧客名】${name} 様
【サービス種別】${type}
【依頼内容】${desc}

以下の形式で出力してください:
---
件名: 【ご見積書】○○のご提案

本文: （300文字程度、丁寧語で）

作業内容:
• 項目1: ○○円
• 項目2: ○○円
合計: ○○円（税込）

備考: （工期・注意事項など）
---
`;

  const response = UrlFetchApp.fetch("https://api.anthropic.com/v1/messages", {
    method: "post",
    headers: {
      "x-api-key": CLAUDE_API_KEY,
      "anthropic-version": "2023-06-01",
      "content-type": "application/json",
    },
    payload: JSON.stringify({
      model: CLAUDE_MODEL,
      max_tokens: 1024,
      messages: [{ role: "user", content: prompt }],
    }),
    muteHttpExceptions: true,
  });

  const result = JSON.parse(response.getContentText());
  const estimateText = result.content[0].text;

  // Googleドキュメントに見積書を自動作成
  const doc = DocumentApp.create(`見積書_${name}_${new Date().toLocaleDateString("ja-JP")}`);
  doc.getBody().setText(estimateText);

  // スプレッドシートにドキュメントURLを記録
  sheet.getRange(row, 9).setValue(doc.getUrl());

  SpreadsheetApp.getUi().alert(`✅ 見積書を作成しました！\n${doc.getUrl()}`);
}
