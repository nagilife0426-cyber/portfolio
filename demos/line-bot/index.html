実装コード（Python / Flask）

LINE Messaging APIとFlaskを組み合わせた予約受付Bot。キーワード解析でメニュー・予約・営業時間に自動分岐します。

# LINE Bot - Flask + line-bot-sdk
from flask import Flask, request, abort
from linebot import LineBotApi, WebhookHandler
from linebot.models import (
    MessageEvent, TextMessage, TextSendMessage,
    TemplateSendMessage, ButtonsTemplate, PostbackAction
)

app = Flask(__name__)
line_bot_api = LineBotApi('YOUR_CHANNEL_ACCESS_TOKEN')
handler = WebhookHandler('YOUR_CHANNEL_SECRET')

user_sessions = {}  # 予約セッション管理（本番はRedis等を使用）

@app.route("/callback", methods=['POST'])
def callback():
    signature = request.headers['X-Line-Signature']
    body = request.get_data(as_text=True)
    try:
        handler.handle(body, signature)
    except Exception:
        abort(400)
    return 'OK'

@handler.add(MessageEvent, message=TextMessage)
def handle_message(event):
    user_id = event.source.user_id
    text = event.message.text

    # キーワード解析
    if '予約' in text or 'よやく' in text:
        reply = create_reservation_menu()
    elif 'キャンセル' in text:
        reply = TextSendMessage(
            text="キャンセルのご希望ですね。\n予約の日時をお知らせください。\n担当者が確認後ご連絡いたします📞"
        )
    elif '料金' in text or '値段' in text:
        reply = create_price_list()
    elif '営業' in text or '時間' in text:
        reply = TextSendMessage(
            text="営業時間のご案内🕐\n\n月〜土: 10:00〜20:00\n日・祝: 10:00〜18:00\n定休日: 毎週火曜日\n\nオンライン予約は24時間受付中です！"
        )
    else:
        reply = create_main_menu()

    line_bot_api.reply_message(event.reply_token, reply)

def create_main_menu():
    return TemplateSendMessage(
        alt_text='メニュー',
        template=ButtonsTemplate(
            title='美容室サクラ 公式LINE',
            text='ご用件をお選びください',
            actions=[
                PostbackAction(label='📅 予約する', data='action=reserve'),
                PostbackAction(label='💰 料金確認', data='action=price'),
                PostbackAction(label='🕐 営業時間', data='action=hours'),
                PostbackAction(label='📍 アクセス', data='action=access'),
            ]
        )
    )
