import requests
from bs4 import BeautifulSoup
import time
from datetime import datetime

def monitor_price(url, selector, interval_minutes=60):
    history = []

    while True:
        res = requests.get(url, headers={'User-Agent': 'Mozilla/5.0'})
        soup = BeautifulSoup(res.text, 'html.parser')
        price_text = soup.select_one(selector).text.strip()
        price = int(price_text.replace('¥','').replace(',',''))

        record = {
            'timestamp': datetime.now().isoformat(),
            'price': price
        }
        history.append(record)

        # 値下がり検知
        if len(history) > 1 and price < history[-2]['price']:
            print(f"値下がり検知: ¥{history[-2]['price']} → ¥{price}")
            # LINE / メール通知をここで呼ぶ
            send_line_notify(price)

        time.sleep(interval_minutes * 60)

def send_line_notify(price):
    requests.post(
        'https://notify-api.line.me/api/notify',
        headers={'Authorization': f'Bearer {LINE_TOKEN}'},
        data={'message': f'値下がり！現在価格: ¥{price:,}'}
    )
