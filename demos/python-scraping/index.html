💡 なぜ Claude API が他と違うのか
SP-API や Keepa は「データを取る」ツールです。Claude API は「データから判断・文章・戦略を生成する」ツール。商品説明の最適化・レビュー分析・アラートの自然言語解説など、AIならではの付加価値を追加できます。Python + SP-API + Claude API の組み合わせが、今の Amazon EC 自動化の最先端です。

amazon_sp_api.py — 売上・在庫データ取得（SP-API） 
# Amazon SP-API を使った売上・在庫データ取得
# スクレイピング不使用 — 公式APIのみ使用（規約準拠）
import boto3
from sp_api.api import Orders, Inventories, Reports
from sp_api.base import Marketplaces, SellingApiException
from datetime import datetime, timedelta
import gspread  # Googleシート連携

# SP-API 認証情報（クライアントから提供いただく情報）
CREDENTIALS = {
    "refresh_token": "YOUR_REFRESH_TOKEN",   # セラーセントラルから取得
    "lwa_app_id":     "YOUR_APP_ID",
    "lwa_client_secret": "YOUR_SECRET",
}

def get_monthly_sales() -> list[dict]:
    """先月の注文データを取得して売上・粗利を集計"""
    orders_api = Orders(credentials=CREDENTIALS, marketplace=Marketplaces.JP)

    end   = datetime.utcnow()
    start = end - timedelta(days=30)

    res = orders_api.get_orders(
        CreatedAfter=start.isoformat() + "Z",
        CreatedBefore=end.isoformat() + "Z",
        OrderStatuses=["Shipped", "Delivered"]
    )
    orders = res.payload.get("Orders", [])
    print(f"取得した注文数: {len(orders)}")
    return orders

def get_inventory_levels() -> list[dict]:
    """全商品の在庫数・補充判定を取得"""
    inv_api = Inventories(credentials=CREDENTIALS, marketplace=Marketplaces.JP)
    res = inv_api.get_inventory_summary_marketplace(
        details=True, marketplaceIds=["A1VC38T7YXB528"]  # JP
    )
    summaries = res.payload.get("inventorySummaries", [])

    # 在庫切れリスク判定（14日分を下回ったらアラート）
    alerts = []
    for item in summaries:
        qty = item.get("totalQuantity", 0)
        daily_velocity = item.get("salesRank", 1) # 実際は売上履歴から算出
        days_left = qty / max(daily_velocity, 0.1)
        if days_left < 14:
            alerts.append({"asin": item["asin"], "days_left": round(days_left)})
    return alerts

def export_to_sheets(data: list[dict]):
    """売上データをGoogleスプレッドシートへ自動出力"""
    gc = gspread.service_account(filename="credentials.json")
    sh = gc.open("Amazon EC 売上管理")
    ws = sh.worksheet("月次データ")

    rows = [[d["asin"], d["sales"], d["profit_rate"], d["acos"]] for d in data]
    ws.append_rows(rows, value_input_option="USER_ENTERED")
    print("✅ Googleシートへの出力完了")

# cron: 毎朝9時に自動実行
# 0 9 * * * python3 amazon_sp_api.py
if __name__ == "__main__":
    sales = get_monthly_sales()
    stock_alerts = get_inventory_levels()
    export_to_sheets(sales)
keepa_monitor.py — 競合価格監視（Keepa API） 
# Keepa APIを使った競合価格監視 + Slack自動通知
import keepa
import requests

KEEPA_API_KEY = "YOUR_KEEPA_KEY"  # Keepa有料プランが必要
SLACK_WEBHOOK  = "YOUR_SLACK_WEBHOOK_URL"

# 監視対象の競合ASIN（クライアントから提供いただく）
COMPETITOR_ASINS = [
    "B09XXXXX01",  # 競合A: ドッグフード 2kg
    "B09XXXXX02",  # 競合B: 犬用おやつ
]

def check_competitor_prices(our_prices: dict) -> list[dict]:
    """競合価格を取得して自社との差分を検出"""
    api = keepa.Keepa(KEEPA_API_KEY)
    products = api.query(COMPETITOR_ASINS, stats=90)

    alerts = []
    for p in products:
        asin = p["asin"]
        # Keepaの価格はセント単位なので100で割る
        current_price = p["stats"]["current"][0] / 100
        our_price = our_prices.get(asin, 0)
        diff = our_price - current_price

        if diff > 200:  # 競合より200円以上高い場合はアラート
            alerts.append({
                "asin": asin,
                "our_price": our_price,
                "competitor_price": current_price,
                "diff": diff,
                "30day_low": p["stats"]["min"][0] / 100,  # 過去30日最安値
            })
    return alerts

def notify_slack(alerts: list[dict]):
    """価格乖離をSlackに通知"""
    if not alerts:
        return

    text = "🚨 *競合価格アラート*\n\n"
    for a in alerts:
        text += (
            f"• ASIN: `{a['asin']}`\n"
            f"  自社: ¥{a['our_price']:,} → 競合: ¥{a['competitor_price']:,} "
            f"（*¥{a['diff']:,} 高い*）\n"
            f"  過去30日最安値: ¥{a['30day_low']:,}\n\n"
        )

    requests.post(SLACK_WEBHOOK, json={"text": text})
    print(f"✅ Slack通知送信: {len(alerts)}件")
ai_listing_optimizer.py — 商品説明文 AI 自動生成（Claude API） 
# Claude API を使った商品タイトル・バレットポイントの自動最適化
# SP-APIで現在のリスティングを取得 → Claude APIで改善案を生成 → 確認後に更新
import anthropic
from sp_api.api import CatalogItems, Listings
from sp_api.base import Marketplaces

CLAUDE_CLIENT = anthropic.Anthropic()  # ANTHROPIC_API_KEY を環境変数にセット

CREDENTIALS = {
    "refresh_token": "YOUR_REFRESH_TOKEN",
    "lwa_app_id": "YOUR_APP_ID",
    "lwa_client_secret": "YOUR_SECRET",
}

def get_current_listing(asin: str) -> dict:
    """SP-APIで現在の商品情報を取得"""
    cat = CatalogItems(credentials=CREDENTIALS, marketplace=Marketplaces.JP)
    res = cat.get_catalog_item(asin=asin, includedData=["summaries","attributes"])
    item = res.payload
    return {
        "title": item["summaries"][0].get("itemName", ""),
        "bullets": item.get("attributes", {}).get("bullet_point", []),
    }

def generate_optimized_listing(current: dict, keywords: list[str], product_info: str) -> dict:
    """Claude APIで最適化されたタイトル・バレットポイントを生成"""
    prompt = f"""
あなたはAmazon日本のSEO専門家です。以下の商品リスティングを改善してください。

【現在のタイトル】
{current['title']}

【バレットポイント】
{chr(10).join(f'• {b}' for b in current['bullets'])}

【ターゲットキーワード】
{', '.join(keywords)}

【商品の特徴・USP】
{product_info}

以下の形式でJSON出力してください:
{{
  "title": "改善後タイトル（80文字以内）",
  "bullets": ["バレット1（最重要メリット）", "バレット2", "バレット3", "バレット4", "バレット5"],
  "改善ポイント": ["変更点の説明"]
}}
"""

    response = CLAUDE_CLIENT.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[{"role": "user", "content": prompt}]
    )
    import json
    return json.loads(response.content[0].text)

def update_listing_with_approval(asin: str, seller_id: str, optimized: dict):
    """人間の確認後にSP-APIでリスティングを更新"""
    print("=== 改善案 ===")
    print(f"タイトル: {optimized['title']}")
    print(f"改善ポイント: {optimized['改善ポイント']}")

    confirm = input("\n反映しますか? (y/N): ").strip().lower()
    if confirm != "y":
        print("キャンセルしました")
        return

    listings = Listings(credentials=CREDENTIALS, marketplace=Marketplaces.JP)
    listings.put_listings_item(
        sellerId=seller_id, sku=asin,
        body={"productType": "PET_FOOD", "attributes": {
            "item_name": [{"value": optimized["title"], "language_tag": "ja_JP"}],
            "bullet_point": [{"value": b, "language_tag": "ja_JP"} for b in optimized["bullets"]],
        }}
    )
    print("✅ リスティングを更新しました")

if __name__ == "__main__":
    asin = "B09PET0001"
    current = get_current_listing(asin)
    optimized = generate_optimized_listing(
        current=current,
        keywords=["プレミアムドッグフード", "無添加 犬 フード", "国産 ドッグフード"],
        product_info="無添加・国産・グレインフリー。シニア犬にも対応。獣医師監修。"
    )
    update_listing_with_approval(asin, "YOUR_SELLER_ID", optimized)
