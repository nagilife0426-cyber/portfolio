実装コード（Python）

requests＋BeautifulSoupで複数サイトを巡回し、価格・在庫・評価を自動収集。毎朝9時にscheduleで実行されます。

import requests
from bs4 import BeautifulSoup
import pandas as pd
import time, schedule
from datetime import datetime

class ProductScraper:
    """複数ECサイトの価格を定期収集するスクレイパー"""

    def __init__(self):
        self.headers = {'User-Agent': 'Mozilla/5.0 ...'}
        self.results = []

    def scrape_amazon(self, url: str) -> dict:
        """Amazon商品ページから価格・評価を取得"""
        try:
            time.sleep(1)  # レート制限対策
            res = requests.get(url, headers=self.headers, timeout=10)
            soup = BeautifulSoup(res.content, 'html.parser')

            price_elem = soup.select_one('.a-price-whole')
            rating_elem = soup.select_one('.a-icon-alt')
            title_elem = soup.select_one('#productTitle')

            return {
                'site': 'Amazon',
                'title': title_elem.text.strip() if title_elem else 'N/A',
                'price': int(price_elem.text.replace(',', '')) if price_elem else None,
                'rating': float(rating_elem.text[:3]) if rating_elem else None,
                'scraped_at': datetime.now().isoformat()
            }
        except Exception as e:
            print(f"エラー: {e}")
            return None

    def scrape_rakuten(self, keyword: str) -> list:
        """楽天APIで最安値を取得"""
        params = {
            'applicationId': 'YOUR_APP_ID',
            'keyword': keyword,
            'hits': 5,
            'sort': '+itemPrice'
        }
        res = requests.get(
            'https://app.rakuten.co.jp/services/api/IchibaItem/Search/20170706',
            params=params
        )
        data = res.json()
        return [{
            'site': '楽天',
            'title': item['Item']['itemName'],
            'price': item['Item']['itemPrice'],
            'rating': item['Item']['reviewAverage'],
        } for item in data.get('Items', [])[:3]]

    def run_daily(self):
        """毎朝9時に自動実行"""
        print(f"スクレイピング開始: {datetime.now()}")
        products = self.load_watch_list('watch_list.csv')

        for product in products:
            amazon = self.scrape_amazon(product['amazon_url'])
            rakuten = self.scrape_rakuten(product['keyword'])
            if amazon: self.results.append(amazon)
            self.results.extend(rakuten)

        df = pd.DataFrame(self.results)
        filename = f"results_{datetime.now():%Y%m%d}.csv"
        df.to_csv(filename, index=False, encoding='utf-8-sig')
        print(f"完了: {len(self.results)}件 → {filename}")

# 毎日9:00に自動実行
scraper = ProductScraper()
schedule.every().day.at("09:00").do(scraper.run_daily)

while True:
    schedule.run_pending()
    time.sleep(60)

注意事項
・各サイトの利用規約を確認してから実行してください
・レート制限（time.sleep）を必ず設定し、サーバーへの負荷を避けること
・楽天/Yahoo等はAPIキーの取得が必要（無料枠あり）
