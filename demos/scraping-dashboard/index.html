実装コード

本番ではcronで定期実行し、結果をDBまたはGoogle Spreadsheetsに保存します。

price_monitor.py Python 3.10+ / requests + BeautifulSoup4 
import requests
from bs4 import BeautifulSoup
import pandas as pd
from datetime import datetime
import time

class PriceMonitor:
    def __init__(self, targets):
        # targets = [{'name': '商品名', 'url': 'URL', 'selector': 'CSSセレクタ'}]
        self.targets = targets
        self.results = []

    def scrape(self):
        for target in self.targets:
            try:
                res = requests.get(target['url'],
                                   headers={'User-Agent': 'Mozilla/5.0'},
                                   timeout=10)
                soup = BeautifulSoup(res.text, 'html.parser')
                price_el = soup.select_one(target['selector'])

                if price_el:
                    price = int(price_el.text.strip()
                                .replace('¥','').replace(',',''))
                    self.results.append({
                        'name': target['name'],
                        'url': target['url'],
                        'price': price,
                        'scraped_at': datetime.now().isoformat()
                    })
            except Exception as e:
                print(f"エラー ({target['name']}): {e}")

            time.sleep(1)  # サーバー負荷軽減

        return pd.DataFrame(self.results)

    def export_csv(self, path='prices.csv'):
        df = self.scrape()
        df.to_csv(path, index=False, encoding='utf-8-sig')
        print(f"保存完了: {path}")
