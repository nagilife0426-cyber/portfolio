🐍 コードサンプル（Claude API × Instagram 自動投稿）
Python 3.11+ 
# sns_auto_post.py - ペットEC向けSNS自動投稿スクリプト
import anthropic, requests, schedule
from datetime import datetime

client = anthropic.Anthropic()

def generate_caption(product_name, platform):
    """Claude APIでSNSキャプションを生成"""
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=500,
        messages=[{
            "role": "user",
            "content": f"ペットEC「onedog Deli」の{product_name}について"
                       f"{platform}向けのキャプションを生成してください。"
                       "感情的に訴え、行動を促す内容で。ハッシュタグも含めて。"
        }]
    )
    return message.content[0].text

def post_to_instagram(caption, image_url):
    """Meta Graph APIで自動投稿"""
    # 画像コンテナ作成 → メディア投稿の2ステップ
    container = requests.post(
        f"https://graph.facebook.com/v19.0/{IG_USER_ID}/media",
        params={"image_url": image_url, "caption": caption,
                "access_token": IG_ACCESS_TOKEN}
    )
    return requests.post(
        f"https://graph.facebook.com/v19.0/{IG_USER_ID}/media_publish",
        params={"creation_id": container.json()["id"],
                "access_token": IG_ACCESS_TOKEN}
    ).json()
